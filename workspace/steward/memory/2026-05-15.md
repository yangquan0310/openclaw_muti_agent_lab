import zipfile
import shutil
import re

v16_path = '/root/仓库/维护老板信息/uploads/杨权-华中师范大学-心理学_v16.docx'
unpack_dir = '/root/仓库/维护老板信息/temp/unpacked_v18'

shutil.rmtree(unpack_dir, ignore_errors=True)
with zipfile.ZipFile(v16_path, 'r') as z:
    z.extractall(unpack_dir)

xml_path = unpack_dir + '/word/document.xml'
with open(xml_path, 'r', encoding='utf-8') as f:
    content = f.read()

# Find the merged table
tables = re.findall(r'<w:tbl[^>]*>.*?</w:tbl>', content, re.DOTALL)
if len(tables) >= 1:
    tbl = tables[0]
    
    # Get all rows
    rows = re.findall(r'<w:tr[^>]*>.*?</w:tr>', tbl, re.DOTALL)
    print(f"Total rows: {len(rows)}")
    
    # The first 5 rows are from table 1 (keep as is)
    # Rows 6+ are from table 2 - need to fix widths
    
    fixed_rows = []
    for i, row in enumerate(rows):
        cells = re.findall(r'<w:tc[^>]*>.*?</w:tc>', row, re.DOTALL)
        
        if i >= 5 and len(cells) > 1:
            # This is a table 2 row with multiple cells - convert to single cell
            # Extract all text from cells
            all_text_parts = []
            for cell in cells:
                cell_text = ''
                for t_match in re.findall(r'<w:t[^>]*>([^<]*)</w:t>', cell):
                    cell_text += t_match
                if cell_text.strip():
                    all_text_parts.append(cell_text)
            
            combined_text = ' '.join(all_text_parts)
            
            # Get row properties (w:trPr)
            trPr_match = re.search(r'<w:trPr>.*?</w:trPr>', row, re.DOTALL)
            trPr = trPr_match.group() if trPr_match else ''
            
            # Get paragraph properties from first cell's paragraph
            first_cell = cells[0]
            pPr_match = re.search(r'<w:pPr>.*?</w:pPr>', first_cell, re.DOTALL)
            pPr = pPr_match.group() if pPr_match else '<w:pPr><w:spacing w:after="0" /></w:pPr>'
            
            # Get run properties from first cell
            rPr_match = re.search(r'<w:rPr>.*?</w:rPr>', first_cell, re.DOTALL)
            if rPr_match:
                rPr = rPr_match.group()
            else:
                rPr = '<w:rPr><w:rFonts w:ascii="微软雅黑" w:hAnsi="微软雅黑" w:cs="微软雅黑" /><w:color w:val="545454" /><w:spacing w:val="-1" /><w:sz w:val="18" /><w:szCs w:val="18" /></w:rPr>'
            
            # Build new single-cell row
            new_row = f'''<w:tr w:rsidR="00474C00" w14:paraId="77777777" w14:textId="77777777" w:rsidTr="00B36AE7">
{trPr}
<w:tc>
<w:tcPr>
<w:tcW w:w="10773" w:type="dxa" />
<w:gridSpan w:val="5" />
</w:tcPr>
<w:p w14:paraId="77777777" w14:textId="77777777" w:rsidR="00474C00" w:rsidRDefault="00474C00" w:rsidP="00474C00">
{pPr}
<w:r w:rsidRPr="00474C00">
{rPr}
<w:t>{combined_text}</w:t>
</w:r>
</w:p>
</w:tc>
</w:tr>'''
            fixed_rows.append(new_row)
        else:
            fixed_rows.append(row)
    
    # Replace table with fixed rows
    new_tbl_content = '\n'.join(fixed_rows)
    
    # Replace in content
    old_tbl = tbl
    # Keep table properties
    tblPr_match = re.search(r'<w:tblPr>.*?</w:tblPr>', tbl, re.DOTALL)
    tblPr = tblPr_match.group() if tblPr_match else '<w:tblPr><w:tblW w:w="0" w:type="auto" /></w:tblPr>'
    
    tblGrid_match = re.search(r'<w:tblGrid>.*?</w:tblGrid>', tbl, re.DOTALL)
    tblGrid = tblGrid_match.group() if tblGrid_match else '<w:tblGrid><w:gridCol w:w="10773" /></w:tblGrid>'
    
    new_tbl = f'<w:tbl>\n{tblPr}\n{new_tbl_content}\n{tblGrid}\n</w:tbl>'
    content = content.replace(old_tbl, new_tbl, 1)
    
    print("Fixed table 2 rows to single column with gridSpan=5")

with open(xml_path, 'w', encoding='utf-8') as f:
    f.write(content)

# Repack
v18_path = '/root/仓库/维护老板信息/uploads/杨权-华中师范大学-心理学_v18.docx'
shutil.make_archive(v18_path.replace('.docx', ''), 'zip', unpack_dir)
shutil.move(v18_path.replace('.docx', '') + '.zip', v18_path)

print(f"SUCCESS: Created {v18_path}")
